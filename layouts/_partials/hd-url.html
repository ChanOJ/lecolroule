{{- /* URL of the high-definition version of an image of assets/images/ (its "_hd" file, at most 2560 px), or "" when it has none. Usage: {{ partial "hd-url.html" "/images/salon.jpg" }} */ -}} {{- $path := strings.TrimPrefix "/" . -}} {{- $ext := path.Ext $path -}} {{- $url := "" -}} {{- with resources.Get (printf "%s_hd%s" (strings.TrimSuffix $ext $path) $ext) -}} {{- /* At most 2560 px (enough for a full screen), JPEG quality 82: the original files would be slow on phones. */ -}} {{- $hd := . -}} {{- if or (gt .Width 2560) (gt .Height 2560) }}{{ $hd = .Fit "2560x2560 q82" }}{{ end -}} {{- $url = ($hd | fingerprint "md5").RelPermalink -}} {{- end -}} {{- return $url -}}
