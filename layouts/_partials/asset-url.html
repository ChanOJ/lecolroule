{{- /* URL of a file of assets/ with a content fingerprint in its name, e.g. "/images/salon.jpg" -> "/images/salon..jpg". The name changes when the file changes, so browsers and the CDN can cache these files for a long time. Usage: {{ partial "asset-url.html" "/images/salon.jpg" }} */ -}} {{- $path := strings.TrimPrefix "/" . -}} {{- with resources.Get $path -}} {{- return (. | fingerprint "md5").RelPermalink -}} {{- else -}} {{- errorf "asset %q not found in assets/" $path -}} {{- end -}}
