{{- /* of a file of assets/images/ (fingerprinted URL, real width and height so that the page does not jump while loading, native lazy loading). Usage: {{ partial "img.html" (dict "src" "/images/x.jpg" "alt" "..." "eager" false) }} Options: "resize" (Hugo image processing spec, e.g. "240x webp"), "priority" (fetchpriority="high", for an eager image that is the largest of the page), "width" and "height" (required for SVG files, whose size Hugo cannot read), "zoom" (opens the image in the viewer of main.js on click: its "_hd" version when assets/images/_hd. exists, else the image itself), "caption" (shown by the viewer). */ -}} {{- $path := strings.TrimPrefix "/" .src -}} {{- $res := resources.Get $path -}} {{- if not $res }}{{ errorf "image %q not found in assets/" $path }}{{ end -}} {{- with .resize }}{{ $res = $res.Resize . }}{{ end -}} {{- $width := .width -}} {{- $height := .height -}} {{- if not $width }}{{ $width = $res.Width }}{{ $height = $res.Height }}{{ end -}} {{- $zoom := "" -}} {{- if .zoom -}} {{- $ext := path.Ext $path -}} {{- $full := resources.Get (printf "%s_hd%s" (strings.TrimSuffix $ext $path) $ext) | default (resources.Get $path) -}} {{- $zoom = ($full | fingerprint "md5").RelPermalink -}} {{- end -}} {{- with $zoom }}{{ end -}} {{- if $zoom }}{{ end -}} {{- /* */ -}}
