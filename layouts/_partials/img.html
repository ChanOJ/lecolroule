{{- /* of a file of assets/images/ (fingerprinted URL, real width and height so that the page does not jump while loading, native lazy loading). Usage: {{ partial "img.html" (dict "src" "/images/x.jpg" "alt" "..." "eager" false) }} Options: "resize" (Hugo image processing spec, e.g. "240x webp"), "priority" (fetchpriority="high", for an eager image that is the largest of the page), "width" and "height" (required for SVG files, whose size Hugo cannot read), "zoom" (when assets/images/_hd. exists, a click opens that high-definition version in the viewer of main.js; images without one are not clickable and not part of the viewer's navigation), "caption" (shown by the viewer). */ -}} {{- $path := strings.TrimPrefix "/" .src -}} {{- $res := resources.Get $path -}} {{- if not $res }}{{ errorf "image %q not found in assets/" $path }}{{ end -}} {{- with .resize }}{{ $res = $res.Resize . }}{{ end -}} {{- $width := .width -}} {{- $height := .height -}} {{- if not $width }}{{ $width = $res.Width }}{{ $height = $res.Height }}{{ end -}} {{- $zoom := "" -}} {{- if .zoom -}} {{- $ext := path.Ext $path -}} {{- with resources.Get (printf "%s_hd%s" (strings.TrimSuffix $ext $path) $ext) -}} {{- /* At most 2560 px (enough for a full screen), JPEG quality 82: the original files (4096 px, 3-5 MB) would be slow on phones. */ -}} {{- $hd := . -}} {{- if or (gt .Width 2560) (gt .Height 2560) }}{{ $hd = .Fit "2560x2560 q82" }}{{ end -}} {{- $zoom = ($hd | fingerprint "md5").RelPermalink -}} {{- end -}} {{- end -}} {{- with $zoom }}{{ end -}} {{- if $zoom }}{{ end -}} {{- /* */ -}}
