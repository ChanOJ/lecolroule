{{- /* of a file of assets/images/ (fingerprinted URL, real width and height so that the page does not jump while loading, native lazy loading). Usage: {{ partial "img.html" (dict "src" "/images/x.jpg" "alt" "..." "eager" false) }} Options: "resize" (Hugo image processing spec, e.g. "240x webp"), "priority" (fetchpriority="high", for an eager image that is the largest of the page), "width" and "height" (required for SVG files, whose size Hugo cannot read), "zoom" (when assets/images/_hd. exists, a click opens that high-definition version in the viewer of main.js; images without one are not clickable and not part of the viewer's navigation), "caption" (shown by the viewer). A JPEG photo with an "_hd" version is shown as a 1200 px copy of that version (see below). */ -}} {{- $path := strings.TrimPrefix "/" .src -}} {{- $res := resources.Get $path -}} {{- if not $res }}{{ errorf "image %q not found in assets/" $path }}{{ end -}} {{- /* Photos with an "_hd" version: the page shows a 1200 px copy of it (sharp on high-density screens: phones, retina displays), the viewer the HD file; two versions only. Without "_hd": the file itself. */ -}} {{- $ext := path.Ext $path -}} {{- if and (not .resize) (in (slice ".jpg" ".jpeg") $ext) -}} {{- with resources.Get (printf "%s_hd%s" (strings.TrimSuffix $ext $path) $ext) -}} {{- if gt .Width 1200 }}{{ $res = .Resize "1200x q80" }}{{ end -}} {{- end -}} {{- end -}} {{- with .resize }}{{ $res = $res.Resize . }}{{ end -}} {{- $width := .width -}} {{- $height := .height -}} {{- if not $width }}{{ $width = $res.Width }}{{ $height = $res.Height }}{{ end -}} {{- $zoom := cond (.zoom | default false) (partial "hd-url.html" .src) "" -}} {{- with $zoom }}{{ end -}} {{- if $zoom }}{{ end -}} {{- /* */ -}}
