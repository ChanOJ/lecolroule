{{- /* Map of a route, drawn on OpenStreetMap: {{< map route="getting-there" height="350" >}} The route is assets/maps/.json (GeoJSON LineString with its mode, car or foot, its distance and the names of both ends), computed once with OSRM (see README.md). main.js loads Leaflet (assets/vendor/) and draws the map when it scrolls into view. No cookie: no consent needed. Without JavaScript, the link below opens the route on openstreetmap.org. */ -}} {{- $name := .Get "route" -}} {{- $res := resources.Get (printf "maps/%s.json" $name) -}} {{- if not $res }}{{ errorf "map: route %q not found in assets/maps/" $name }}{{ end -}} {{- $route := $res | transform.Unmarshal -}} {{- $p := $route.properties -}} {{- $coords := $route.geometry.coordinates -}} {{- $start := index $coords 0 -}} {{- $end := index $coords (sub (len $coords) 1) -}} {{- $osm := printf "https://www.openstreetmap.org/directions?engine=fossgis_osrm_%s&route=%v,%v;%v,%v" $p.mode (index $start 1) (index $start 0) (index $end 1) (index $end 0) -}} {{- $km := div (float $p.distance) 1000 -}} {{- $distance := cond (lt $p.distance 1000) (printf "%d m" (int $p.distance)) (printf "%s km" (lang.FormatNumber 1 $km)) -}} {{ $p.from }} → {{ $p.to }}, {{ $distance }}{{ if eq $p.mode "foot" }} {{ i18n "mapOnFoot" }}{{ end }} · {{ i18n "mapOpen" }} {{- /* */ -}}
