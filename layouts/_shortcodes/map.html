{{- /* Map of a route, drawn on OpenStreetMap: {{< map route="getting-there" height="350" >}}, {{< map route="walk-trail" gpx="village" >}} The route is assets/maps/.json (GeoJSON LineString with its mode, car or foot, its distance and the names of both ends), computed once with OSRM (see README.md). Below the map, links open the same route (from its first to its last point, by car or on foot) in OpenStreetMap, Google Maps and Apple Maps, to see it large or follow it. Option gpx="": a link below the map downloads assets/maps/.gpx. main.js loads Leaflet (assets/vendor/) and draws the map when it scrolls into view. No cookie: no consent needed. */ -}} {{- $name := .Get "route" -}} {{- $res := resources.Get (printf "maps/%s.json" $name) -}} {{- if not $res }}{{ errorf "map: route %q not found in assets/maps/" $name }}{{ end -}} {{- $route := $res | transform.Unmarshal -}} {{- $p := $route.properties -}} {{- $coords := $route.geometry.coordinates -}} {{- $start := index $coords 0 -}} {{- $end := index $coords (sub (len $coords) 1) -}} {{- $a := printf "%v,%v" (index $start 1) (index $start 0) -}} {{- $b := printf "%v,%v" (index $end 1) (index $end 0) -}} {{- $car := eq $p.mode "car" -}} {{- $links := slice (dict "name" "OpenStreetMap" "url" (printf "https://www.openstreetmap.org/directions?engine=fossgis_osrm_%s&route=%s;%s" $p.mode $a $b)) (dict "name" "Google Maps" "url" (printf "https://www.google.com/maps/dir/?api=1&origin=%s&destination=%s&travelmode=%s" $a $b (cond $car "driving" "walking"))) (dict "name" (i18n "appleMaps") "url" (printf "https://maps.apple.com/?saddr=%s&daddr=%s&dirflg=%s" $a $b (cond $car "d" "w"))) -}}

{{ i18n "mapOpenIn" }} {{- range $i, $l := $links }}{{ if $i }} ·{{ end }} {{ $l.name }}{{ end -}}

{{- with .Get "gpx" }} {{- $gpx := resources.Get (printf "maps/%s.gpx" .) }} {{- if not $gpx }}{{ errorf "map: GPX %q not found in assets/maps/" . }}{{ end }} {{ i18n "mapGpx" }} {{- end }}
